FILE: 20-head-elements.html
======================================================================
<!DOCTYPE html>
<html>
<head>
<!-- The title sets the text displayed in the browser tab. -->
<title>Important Head Elements</title>
<!-- The charset defines the character encoding. -->
<meta charset="UTF-8">
<!-- The description provides a summary of the webpage. -->
<meta name="description" content="A webpage demonstrating important HTML head elements">
<!-- The author identifies who created the webpage. -->
<meta name="author" content="Jayant">
<!-- The base sets the base URL for relative URLs. -->
<base href="https://example.com/">
<!-- The link connects an external resource such as a favicon. -->
<link rel="icon" href="favicon.ico">
<!-- The style tag contains CSS. -->
<style></style>
<!-- The script tag contains JavaScript. -->
<script></script>
</head>
<body>
<h1>Important HTML Head Elements</h1>
<p>This page demonstrates important elements used inside the head section.</p>
</body>
</html>
